<script>
  const totalPreguntas = 5;
  let preguntaActual = -1; // -1 = pantalla inicio

  const pantallaInicio = document.getElementById('pantalla-inicio');
  const pantallaFinal = document.getElementById('pantalla-final');
  const preguntas = Array.from(document.querySelectorAll('.pregunta'))
    .sort((a,b) => a.dataset.pregunta - b.dataset.pregunta);
  const toast = document.getElementById('toast');

  function ocultarTodo(){
    pantallaInicio.classList.remove('active');
    pantallaFinal.classList.remove('active');
    preguntas.forEach(p => p.classList.remove('active'));
  }

  function irAPregunta(indice){
    ocultarTodo();
    if(indice >= totalPreguntas){
      mostrarFinal();
      return;
    }
    preguntaActual = indice;
    preguntas[indice].classList.add('active');
  }

  function mostrarFinal(){
    pantallaFinal.classList.add('active');
    lanzarConfeti();
  }

  function mostrarToast(){
    toast.classList.add('mostrar');
    setTimeout(() => toast.classList.remove('mostrar'), 1600);
  }

  preguntas.forEach((seccion) => {
    const opciones = seccion.querySelectorAll('.opcion');
    opciones.forEach(boton => {
      boton.addEventListener('click', () => {
        const esCorrecta = boton.dataset.correcta === 'true';
        if(esCorrecta){
          boton.classList.add('correcta');
          setTimeout(() => {
            boton.classList.remove('correcta');
            irAPregunta(preguntaActual + 1);
          }, 500);
        } else {
          boton.classList.add('incorrecta');
          mostrarToast();
          setTimeout(() => boton.classList.remove('incorrecta'), 400);
        }
      });
    });
  });

  /* ---- Confeti simple con canvas, sin librerías externas ---- */
  function lanzarConfeti(){
    const canvas = document.getElementById('canvasConfeti');
    const ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;

    const colores = ['#ffb4a2','#e0a458','#8fbf8f','#e76f51','#f4d35e'];
    const particulas = Array.from({length:120}, () => ({
      x: Math.random() * canvas.width,
      y: -20 - Math.random() * canvas.height,
      r: 4 + Math.random() * 5,
      color: colores[Math.floor(Math.random()*colores.length)],
      velY: 2 + Math.random() * 3,
      velX: -1 + Math.random() * 2,
      rot: Math.random() * 360,
      velRot: -4 + Math.random() * 8
    }));

    let frames = 0;
    const maxFrames = 420; // ~7 segundos a 60fps

    function animar(){
      ctx.clearRect(0,0,canvas.width, canvas.height);
      particulas.forEach(p => {
        p.y += p.velY;
        p.x += p.velX;
        p.rot += p.velRot;
        if(p.y > canvas.height + 20) p.y = -20;

        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.rot * Math.PI/180);
        ctx.fillStyle = p.color;
        ctx.fillRect(-p.r/2, -p.r/2, p.r, p.r*1.6);
        ctx.restore();
      });
      frames++;
      if(frames < maxFrames){
        requestAnimationFrame(animar);
      } else {
        ctx.clearRect(0,0,canvas.width, canvas.height);
      }
    }
    animar();
  }

  window.addEventListener('resize', () => {
    const canvas = document.getElementById('canvasConfeti');
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  });
</script>
